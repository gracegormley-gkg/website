Add an inline `<script>` before `</body>` (after `<site-footer>`) that fetches data.json and renders each video:

```javascript
<script>
function isPlaceholder(url) {
  return !url || !url.startsWith('http');
}

function toEmbedUrl(url) {
  const ytMatch = url.match(/(?:youtube\.com\/watch\?v=|youtu\.be\/)([^&?/]+)/);
  if (ytMatch) return 'https://www.youtube-nocookie.com/embed/' + ytMatch[1];
  const viMatch = url.match(/vimeo\.com\/(\d+)/);
  if (viMatch) return 'https://player.vimeo.com/video/' + viMatch[1];
  return url;
}

fetch('./data.json')
  .then(function(r) { return r.json(); })
  .then(function(data) {
    var container = document.getElementById('video-list');
    if (!data.videos || data.videos.length === 0) {
      container.innerHTML = '<p>No videos available.</p>';
      return;
    }
    container.innerHTML = data.videos.map(function(v) {
      var isReal = !isPlaceholder(v.embedUrl);
      var embedSrc = isReal ? toEmbedUrl(v.embedUrl) : '';
      return '<article class="video-item">' +
        '<h2 class="video-title">' + v.title + '</h2>' +
        (isReal
          ? '<div class="video-embed"><iframe src="' + embedSrc + '" title="' + v.title + '" allowfullscreen loading="lazy"></iframe></div>'
          : '<div class="video-placeholder" aria-label="Video coming soon"><span>Video available upon request</span></div>'
        ) +
        (v.description ? '<p class="video-desc">' + v.description + '</p>' : '') +
        '</article>';
    }).join('');
  });
</script>
```

Important details:
- Use `var` not `const`/`let` to avoid any strict-mode or browser quirks on older Safari — though all modern browsers support `const`, using `var` here is consistent with how the existing clips.html inline script was written
- `toEmbedUrl` must handle three input formats: `youtube.com/watch?v=ID`, `youtu.be/ID`, and already-embed URLs — pass-through for anything not matched
- The placeholder div must have visible text ("Video available upon request") so it never looks broken
- `loading="lazy"` on iframe to avoid blocking page load
- Do NOT include `frameborder="0"` on the iframe — use CSS `border: none` in style.css instead (frameborder is deprecated HTML)
